 CUSTOMER SALES INTELLIGENCE
 PostgreSQL Business Analysis

01. DATA VALIDATION


Total number of orders
SELECT COUNT(*) AS total_orders
FROM orders;


 Check duplicate Order IDs
SELECT
    Order_ID,
    COUNT(*) AS duplicate_count
FROM orders
GROUP BY Order_ID
HAVING COUNT(*) > 1;


 Check NULL values
SELECT
    COUNT(*) FILTER (WHERE Order_ID IS NULL) AS missing_order_id,
    COUNT(*) FILTER (WHERE Order_Date IS NULL) AS missing_order_date,
    COUNT(*) FILTER (WHERE Customer_ID IS NULL) AS missing_customer_id,
    COUNT(*) FILTER (WHERE Product_ID IS NULL) AS missing_product_id,
    COUNT(*) FILTER (WHERE Quantity IS NULL) AS missing_quantity,
    COUNT(*) FILTER (WHERE Discount IS NULL) AS missing_discount,
    COUNT(*) FILTER (WHERE Sales IS NULL) AS missing_sales,
    COUNT(*) FILTER (WHERE Profit IS NULL) AS missing_profit,
    COUNT(*) FILTER (WHERE Payment_Method IS NULL) AS missing_payment_method,
    COUNT(*) FILTER (WHERE Order_Status IS NULL) AS missing_order_status
FROM orders;


Check invalid quantity
SELECT *
FROM orders
WHERE Quantity <= 0;


 Check invalid discount
SELECT *
FROM orders
WHERE Discount < 0
   OR Discount > 1;


 Check negative sales
SELECT *
FROM orders
WHERE Sales < 0;


 02. OVERALL BUSINESS KPIs


SELECT
    COUNT(DISTINCT Order_ID) AS total_orders,
    COUNT(DISTINCT Customer_ID) AS total_customers,
    ROUND(SUM(Sales), 2) AS total_sales,
    ROUND(SUM(Profit), 2) AS total_profit,
    ROUND(AVG(Sales), 2) AS average_order_value,
    ROUND(
        SUM(Profit) / NULLIF(SUM(Sales), 0) * 100,
        2
    ) AS profit_margin
FROM orders;



03. MONTHLY SALES ANALYSIS


SELECT
    DATE_TRUNC('month', Order_Date)::date AS month,
    COUNT(DISTINCT Order_ID) AS total_orders,
    ROUND(SUM(Sales), 2) AS total_sales,
    ROUND(SUM(Profit), 2) AS total_profit,
    ROUND(
        SUM(Profit) / NULLIF(SUM(Sales), 0) * 100,
        2
    ) AS profit_margin
FROM orders
GROUP BY 1
ORDER BY 1;


 Highest sales month
SELECT
    DATE_TRUNC('month', Order_Date)::date AS month,
    ROUND(SUM(Sales), 2) AS total_sales
FROM orders
GROUP BY 1
ORDER BY total_sales DESC
LIMIT 1;


 Lowest sales month
SELECT
    DATE_TRUNC('month', Order_Date)::date AS month,
    ROUND(SUM(Sales), 2) AS total_sales
FROM orders
GROUP BY 1
ORDER BY total_sales ASC
LIMIT 1;



04. MONTH-OVER-MONTH SALES GROWTH

WITH monthly_sales AS (
    SELECT
        DATE_TRUNC('month', Order_Date)::date AS month,
        SUM(Sales) AS total_sales
    FROM orders
    GROUP BY 1
)

SELECT
    month,
    ROUND(total_sales, 2) AS total_sales,
    ROUND(
        (
            total_sales
            - LAG(total_sales) OVER (ORDER BY month)
        )
        / NULLIF(
            LAG(total_sales) OVER (ORDER BY month),
            0
        ) * 100,
        2
    ) AS mom_growth_percentage
FROM monthly_sales
ORDER BY month;

        SUM(Profit) / NULLIF(SUM(Sales), 0) * 100,

   AND SUM(Profit) / NULLIF(SUM(Sales), 0) < 0.12

 08. RFM ANALYSIS


WITH customer_metrics AS (
    SELECT
        Customer_ID,
        MAX(Order_Date) AS last_order_date,
        COUNT(DISTINCT Order_ID) AS frequency,
        SUM(Sales) AS monetary
    FROM orders
    GROUP BY Customer_ID
),

rfm_scores AS (
    SELECT
        Customer_ID,
        last_order_date,
        CURRENT_DATE - last_order_date AS recency_days,
        frequency,
        monetary,

        NTILE(5) OVER (
            ORDER BY last_order_date DESC
        ) AS r_score,

        NTILE(5) OVER (
            ORDER BY frequency
        ) AS f_score,

        NTILE(5) OVER (
            ORDER BY monetary
        ) AS m_score

    FROM customer_metrics
)

SELECT
    Customer_ID,
    last_order_date,
    recency_days,
    frequency,
    ROUND(monetary, 2) AS monetary,
    r_score,
    f_score,
    m_score,

    CONCAT(
        r_score,
        f_score,
        m_score
    ) AS rfm_score

FROM rfm_scores
ORDER BY rfm_score DESC;



 09. RFM CUSTOMER SEGMENTATION


WITH customer_metrics AS (
    SELECT
        Customer_ID,
        MAX(Order_Date) AS last_order_date,
        COUNT(DISTINCT Order_ID) AS frequency,
        SUM(Sales) AS monetary
    FROM orders
    GROUP BY Customer_ID
),

rfm_scores AS (
    SELECT
        *,
        CURRENT_DATE - last_order_date AS recency_days,

        NTILE(5) OVER (
            ORDER BY last_order_date DESC
        ) AS r_score,

        NTILE(5) OVER (
            ORDER BY frequency
        ) AS f_score,

        NTILE(5) OVER (
            ORDER BY monetary
        ) AS m_score

    FROM customer_metrics
)

SELECT
    Customer_ID,
    recency_days,
    frequency,
    ROUND(monetary, 2) AS monetary,

    CASE
        WHEN r_score >= 4
         AND f_score >= 4
         AND m_score >= 4
            THEN 'Champions'

        WHEN r_score >= 3
         AND f_score >= 3
            THEN 'Loyal Customers'

        WHEN r_score >= 4
         AND f_score <= 2
            THEN 'New Customers'

        WHEN r_score <= 2
         AND f_score >= 3
            THEN 'At Risk'

        WHEN r_score <= 2
         AND f_score <= 2
            THEN 'Lost Customers'

        ELSE 'Potential Customers'
    END AS customer_segment

FROM rfm_scores;



 10. RFM SEGMENT SUMMARY


WITH customer_metrics AS (
    SELECT
        Customer_ID,
        MAX(Order_Date) AS last_order_date,
        COUNT(DISTINCT Order_ID) AS frequency,
        SUM(Sales) AS monetary
    FROM orders
    GROUP BY Customer_ID
),

rfm_scores AS (
    SELECT
        *,
        CURRENT_DATE - last_order_date AS recency_days,

        NTILE(5) OVER (
            ORDER BY last_order_date DESC
        ) AS r_score,

        NTILE(5) OVER (
            ORDER BY frequency
        ) AS f_score,

        NTILE(5) OVER (
            ORDER BY monetary
        ) AS m_score

    FROM customer_metrics
),

segmented AS (
    SELECT
        *,
        CASE
            WHEN r_score >= 4
             AND f_score >= 4
             AND m_score >= 4
                THEN 'Champions'

            WHEN r_score >= 3
             AND f_score >= 3
                THEN 'Loyal Customers'

            WHEN r_score >= 4
             AND f_score <= 2
                THEN 'New Customers'

            WHEN r_score <= 2
             AND f_score >= 3
                THEN 'At Risk'

            WHEN r_score <= 2
             AND f_score <= 2
                THEN 'Lost Customers'

            ELSE 'Potential Customers'
        END AS customer_segment

    FROM rfm_scores
)

SELECT
    customer_segment,
    COUNT(*) AS customer_count,
    ROUND(SUM(monetary), 2) AS total_sales,
    ROUND(AVG(monetary), 2) AS average_customer_value
FROM segmented
GROUP BY customer_segment
ORDER BY total_sales DESC;



 11. CUSTOMER VALUE SEGMENTATION


WITH customer_value AS (
    SELECT
        Customer_ID,
        COUNT(DISTINCT Order_ID) AS total_orders,
        SUM(Sales) AS total_sales,
        SUM(Profit) AS total_profit
    FROM orders
    GROUP BY Customer_ID
),

segmented AS (
    SELECT
        *,
        NTILE(3) OVER (
            ORDER BY total_sales
        ) AS value_group
    FROM customer_value
)

SELECT
    CASE
        WHEN value_group = 1 THEN 'Low Value'
        WHEN value_group = 2 THEN 'Medium Value'
        WHEN value_group = 3 THEN 'High Value'
    END AS value_segment,

    COUNT(*) AS customer_count,
    SUM(total_orders) AS total_orders,
    ROUND(SUM(total_sales), 2) AS total_sales,
    ROUND(SUM(total_profit), 2) AS total_profit,
    ROUND(
        SUM(total_profit)
        / NULLIF(SUM(total_sales), 0) * 100,
        2
    ) AS profit_margin

FROM segmented
GROUP BY value_group
ORDER BY value_group;

 13. ORDER STATUS ANALYSIS


SELECT
    Order_Status,
    COUNT(DISTINCT Order_ID) AS order_count,
    ROUND(
        COUNT(DISTINCT Order_ID)::numeric
        / SUM(COUNT(DISTINCT Order_ID))
          OVER () * 100,
        2
    ) AS order_percentage
FROM orders
GROUP BY Order_Status
ORDER BY order_count DESC;


-- Order status with sales and profit
SELECT
    Order_Status,
    COUNT(DISTINCT Order_ID) AS order_count,
    ROUND(SUM(Sales), 2) AS total_sales,
    ROUND(SUM(Profit), 2) AS total_profit
FROM orders
GROUP BY Order_Status
ORDER BY order_count DESC;



 14. DELIVERY / CANCELLATION / RETURN RATE


SELECT
    ROUND(
        COUNT(*) FILTER (
            WHERE Order_Status = 'Delivered'
        )::numeric
        / COUNT(*) * 100,
        2
    ) AS delivery_rate,

    ROUND(
        COUNT(*) FILTER (
            WHERE Order_Status = 'Cancelled'
        )::numeric
        / COUNT(*) * 100,
        2
    ) AS cancellation_rate,

    ROUND(
        COUNT(*) FILTER (
            WHERE Order_Status = 'Returned'
        )::numeric
        / COUNT(*) * 100,
        2
    ) AS return_rate

FROM orders;

 17. PRODUCT RANKING USING WINDOW FUNCTION


WITH product_performance AS (
    SELECT
        Product_ID,
        SUM(Sales) AS total_sales,
        SUM(Profit) AS total_profit
    FROM orders
    GROUP BY Product_ID
)

SELECT
    Product_ID,
    ROUND(total_sales, 2) AS total_sales,
    ROUND(total_profit, 2) AS total_profit,

    RANK() OVER (
        ORDER BY total_sales DESC
    ) AS sales_rank,

    RANK() OVER (
        ORDER BY total_profit DESC
    ) AS profit_rank

FROM product_performance
ORDER BY sales_rank;
